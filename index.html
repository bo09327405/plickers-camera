/**
 * 上場點名 — Plickers 卡片影像辨識點名系統（Google Apps Script 後端）
 * 支援前端 GitHub Pages API 串接版
 */

const SPREADSHEET_ID = '1c4AYlKP4lB44Lp33o7BWQO9u8OivycVQ-IhLsWuKdx4';
const LOG_SHEET = '點名紀錄';
const HEADER_ROWS = 2;          // 第 1、2 列為標題
const FIRST_DATE_COL = 3;       // C 欄開始是日期
const STAT_HEADERS = ['加分', '缺席', '遲到', '出席成績'];

/** 
 * API 進入點 (供 GitHub Pages 呼叫)
 * 前端使用 POST 請求，Content-Type 設為 text/plain 避免 CORS 預檢
 */
function doPost(e) {
  const lock = LockService.getScriptLock();
  lock.waitLock(30000);
  try {
    if (!e || !e.postData || !e.postData.contents) {
      throw new Error("無效的請求內容");
    }

    const req = JSON.parse(e.postData.contents);
    const action = req.action;
    let res = {};

    if (action === 'getBootstrap') {
      res = getBootstrap();
    } else if (action === 'getRoster') {
      res = getRoster(req.sheetName, req.iso);
    } else if (action === 'startSession') {
      res = startSession(req.sheetName, req.iso);
    } else if (action === 'recordBatch') {
      res = recordBatch(req.sheetName, req.iso, req.records, req.overwrite);
    } else if (action === 'markAbsent') {
      res = markAbsent(req.sheetName, req.iso, req.rows);
    } else {
      throw new Error('Unknown action: ' + action);
    }

    return ContentService.createTextOutput(JSON.stringify({success: true, data: res}))
      .setMimeType(ContentService.MimeType.JSON);

  } catch(err) {
    return ContentService.createTextOutput(JSON.stringify({success: false, error: err.toString()}))
      .setMimeType(ContentService.MimeType.JSON);
  } finally {
    lock.releaseLock();
  }
}

// 供瀏覽器直接存取測試用
function doGet() {
  return ContentService.createTextOutput("上場點名 API 運作中！").setMimeType(ContentService.MimeType.TEXT);
}

function ss_() { return SpreadsheetApp.openById(SPREADSHEET_ID); }

/** 判斷工作表是否為班級名單（第 2 列有「姓名」） */
function isClassSheet_(sh) {
  if (sh.getName() === LOG_SHEET) return false;
  if (sh.getLastRow() < HEADER_ROWS) return false;
  const r2 = sh.getRange(2, 1, 1, 2).getDisplayValues()[0];
  return String(r2[1]).trim() === '姓名';
}

/** 解析 B 欄「01伍少宇」→ { seat: 1, name: '伍少宇' } */
function parseStudent_(text) {
  const m = String(text).trim().match(/^(\d{1,2})\s*[\.\-、]?\s*(.+)$/);
  if (!m) return null;
  return { seat: parseInt(m[1], 10), name: m[2].trim() };
}

/** 從第 1 列找統計區開始的欄（「加分」），找不到則用最後一欄 + 1 */
function statStartCol_(sh) {
  const lastCol = sh.getLastColumn();
  const r1 = sh.getRange(1, 1, 1, lastCol).getDisplayValues()[0];
  for (let c = FIRST_DATE_COL - 1; c < r1.length; c++) {
    if (STAT_HEADERS.indexOf(String(r1[c]).trim()) >= 0) return c + 1;
  }
  return lastCol + 1;
}

/** '2026-09-07' → 'M/d' 比對字串 '9/7' */
function mdKey_(iso) {
  const p = iso.split('-');
  return parseInt(p[1], 10) + '/' + parseInt(p[2], 10);
}

function cellMd_(display) {
  const m = String(display).trim().match(/(\d{1,2})[\/\-.](\d{1,2})$/);
  return m ? parseInt(m[1], 10) + '/' + parseInt(m[2], 10) : '';
}

/** 找日期欄；create=true 時若不存在就在日期區最後一個空欄建立 */
function findDateCol_(sh, iso, create) {
  const key = mdKey_(iso);
  const endCol = statStartCol_(sh) - 1; // 日期區最後一欄
  if (endCol < FIRST_DATE_COL) return { col: 0, created: false };
  const r2 = sh.getRange(2, FIRST_DATE_COL, 1, endCol - FIRST_DATE_COL + 1).getDisplayValues()[0];
  let lastFilled = -1;
  for (let i = 0; i < r2.length; i++) {
    if (cellMd_(r2[i]) === key) return { col: FIRST_DATE_COL + i, created: false };
    if (String(r2[i]).trim() !== '') lastFilled = i;
  }
  if (!create) return { col: 0, created: false };

  let col;
  if (lastFilled + 1 < r2.length) {
    col = FIRST_DATE_COL + lastFilled + 1;           // 最後一個日期後面的空欄
  } else {
    sh.insertColumnBefore(endCol + 1);                // 沒空間了：在統計區前插入一欄
    col = endCol + 1;
  }
  sh.getRange(2, col).setNumberFormat('@').setValue(key);
  return { col: col, created: true };
}

/** 前端初始化：班級清單＋上次設定 */
function getBootstrap() {
  const classes = ss_().getSheets().filter(isClassSheet_).map(sh => {
    const label = String(sh.getRange(1, 1).getDisplayValue() || sh.getName()).trim();
    const names = sh.getRange(HEADER_ROWS + 1, 2, Math.max(1, sh.getLastRow() - HEADER_ROWS), 1).getDisplayValues();
    const count = names.filter(r => parseStudent_(r[0])).length;
    return { sheet: sh.getName(), label: label, count: count };
  });
  return { classes: classes, tz: Session.getScriptTimeZone() };
}

/** 讀取班級名單（以及該日期既有的紀錄） */
function getRoster(sheetName, iso) {
  const sh = ss_().getSheetByName(sheetName);
  if (!sh) throw new Error('找不到工作表「' + sheetName + '」');
  const n = Math.max(0, sh.getLastRow() - HEADER_ROWS);
  const names = n ? sh.getRange(HEADER_ROWS + 1, 2, n, 1).getDisplayValues() : [];
  const dc = findDateCol_(sh, iso, false);
  const marks = dc.col && n ? sh.getRange(HEADER_ROWS + 1, dc.col, n, 1).getDisplayValues() : [];
  const students = [];
  names.forEach((r, i) => {
    const st = parseStudent_(r[0]);
    if (st) students.push({ seat: st.seat, name: st.name, row: HEADER_ROWS + 1 + i, mark: marks[i] ? String(marks[i][0]).trim() : '' });
  });
  return {
    sheet: sheetName,
    label: String(sh.getRange(1, 1).getDisplayValue() || sheetName).trim(),
    dateKey: mdKey_(iso),
    dateCol: dc.col,
    students: students
  };
}

/** 開始點名：確保日期欄存在 */
function startSession(sheetName, iso) {
  const sh = ss_().getSheetByName(sheetName);
  const dc = findDateCol_(sh, iso, true);
  const roster = getRoster(sheetName, iso);
  roster.created = dc.created;
  roster.colLetter = colLetter_(dc.col);
  return roster;
}

/**
 * 批次寫入點名結果
 * records: [{ row, seat, name, status: '1'|'L'|'x', time: 'HH:mm:ss', card }]
 * overwrite=false：儲存格已有內容就不覆蓋
 */
function recordBatch(sheetName, iso, records, overwrite) {
  const book = ss_();
  const sh = book.getSheetByName(sheetName);
  const dc = findDateCol_(sh, iso, true);
  const results = [];
  const logRows = [];
  records.forEach(rec => {
    const cell = sh.getRange(rec.row, dc.col);
    const cur = String(cell.getDisplayValue()).trim();
    // 核對該列確實是這位學生
    const st = parseStudent_(sh.getRange(rec.row, 2).getDisplayValue());
    if (!st || st.seat !== rec.seat) {
      results.push({ row: rec.row, ok: false, reason: '名單列號已變動，請重新載入' });
      return;
    }
    if (cur !== '' && !overwrite) {
      results.push({ row: rec.row, ok: true, kept: true, mark: cur });
      return;
    }
    if (rec.status === '') cell.clearContent();
    else cell.setValue(rec.status === '1' ? 1 : rec.status);
    results.push({ row: rec.row, ok: true, mark: rec.status });
    logRows.push([new Date(), sh.getName(), mdKey_(iso), rec.seat, st.name, labelOf_(rec.status), rec.time || '', rec.card || '', overwrite ? '手動' : '卡片']);
  });
  if (logRows.length) appendLog_(book, logRows);
  return { dateCol: dc.col, results: results };
}

/** 結束點名：未到的人寫入 x */
function markAbsent(sheetName, iso, rows) {
  const recs = rows.map(r => ({ row: r.row, seat: r.seat, status: 'x', time: '', card: '' }));
  return recordBatch(sheetName, iso, recs, false);
}

function labelOf_(s) { return s === '1' ? '準時' : s === 'L' ? '遲到' : s === 'x' ? '缺席' : '清除'; }

function appendLog_(book, rows) {
  let log = book.getSheetByName(LOG_SHEET);
  if (!log) {
    log = book.insertSheet(LOG_SHEET);
    log.appendRow(['寫入時間', '班級', '日期', '座號', '姓名', '狀態', '辨識時間', '卡號', '方式']);
    log.setFrozenRows(1);
    log.getRange('A:A').setNumberFormat('yyyy/MM/dd HH:mm:ss');
  }
  log.getRange(log.getLastRow() + 1, 1, rows.length, rows[0].length).setValues(rows);
}

function colLetter_(c) {
  let s = '';
  while (c > 0) { const m = (c - 1) % 26; s = String.fromCharCode(65 + m) + s; c = (c - m - 1) / 26; }
  return s;
}
